// ============================================================================ // PROJECT: ESTIQATSY BOT - MINI APP BACKEND API (VERSIONE RESILIENTE) // FILE: Modulo_WebApp_API.gs // ============================================================================ function handleWebAppGetRequest(e) { var params = e ? e.parameter : {}; var action = (params.action || "ping").toLowerCase(); var initDataRaw = params.initData || ""; function jsonResponse(data, status) { var output = { status: status || 200, timestamp: new Date().toISOString(), data: data }; return ContentService.createTextOutput(JSON.stringify(output)) .setMimeType(ContentService.MimeType.JSON); } if (action === "ping") { return jsonResponse({ message: "Estiqatsy WebApp API Online OK" }); } // Estrazione identità utente da Telegram var targetChatId = ""; var userNome = "Ospite"; var userUsername = "@anon"; // 1. Tenta decodifica parametri Telegram WebApp if (initDataRaw) { try { var parts = initDataRaw.split("&"); for (var i = 0; i < parts.length; i++) { var kv = parts[i].split("="); if (decodeURIComponent(kv[0]) === "user") { var uObj = JSON.parse(decodeURIComponent(kv.slice(1).join("="))); targetChatId = String(uObj.id || ""); userNome = uObj.first_name || "Avventuriero"; userUsername = uObj.username ? "@" + uObj.username : "@anon"; break; } } } catch (eErr) { Logger.log("Errore parsing user initData: " + eErr); } } // Fallback se avviato da browser o parametro di test if (!targetChatId && params.chat_id) { targetChatId = String(params.chat_id); } // 1. PROFILO UTENTE & STATO if (action === "profile") { var userObj = null; if (targetChatId && typeof getOrCreateUser === "function") { userObj = getOrCreateUser({ id: targetChatId, first_name: userNome, username: userUsername.replace("@", "") }, "MiniApp"); } return jsonResponse({ chatId: targetChatId || "12345678", nome: userObj ? userObj.nome : userNome, username: userObj ? userObj.username : userUsername, saldoMegoin: userObj ? userObj.megoin : 100, puntiFedelta: userObj ? userObj.punti : 1, piano: userObj ? userObj.plan : "Free" }); } // 2. CATALOGO SHOP if (action === "shop") { var shopItems = []; var shShop = DB.getShop(); if (shShop) { var rows = shShop.getDataRange().getDisplayValues(); for (var i = 1; i < rows.length; i++) { if (rows[i][0] && rows[i][0] !== "—") { shopItems.push({ id: rows[i][0], categoria: rows[i][2] || "Generale", nome: rows[i][4] || rows[i][0], descrizione: rows[i][5] || "", prezzoEuro: rows[i][8] || "0", prezzoMegoin: rows[i][10] || "0", mediaUrl: rows[i][12] || "" }); } } } return jsonResponse({ items: shopItems }); } // 3. CATALOGO RICETTE if (action === "recipes") { var recipeItems = []; var shRecipes = DB.getRecipes(); if (shRecipes) { var rRows = shRecipes.getDataRange().getDisplayValues(); for (var r = 1; r < rRows.length; r++) { if (rRows[r][0] && rRows[r][0] !== "—") { recipeItems.push({ id: r, piatto: rRows[r][0], categoria: rRows[r][1] || "Generale", costo: rRows[r][4] || "Economico", tempo: rRows[r][6] || "15 min", mediaUrl: rRows[r][12] || "" }); } } } return jsonResponse({ recipes: recipeItems }); } // 4. ELENCO GIOCHI if (action === "games") { var gamesData = []; if (typeof getRegisteredGames === "function" && typeof getEpisodesInTab === "function") { var reg = getRegisteredGames(); for (var gk in reg) { var eps = getEpisodesInTab(gk); var epList = Object.keys(eps).map(function(k) { return eps[k]; }); gamesData.push({ gameKey: gk, titolo: reg[gk].tabName.replace("🎮", "").trim(), episodi: epList }); } } return jsonResponse({ games: gamesData }); } return jsonResponse({ error: "Comando non riconosciuto" }, 404); }
